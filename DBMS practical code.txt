DELIMITER $$

CREATE PROCEDURE ShowIssuedBooks()
BEGIN
    DECLARE done INT DEFAULT 0;
    DECLARE b_id INT;
    DECLARE s_name VARCHAR(100);
    DECLARE i_date DATE;


    DECLARE cur CURSOR FOR
        SELECT BookID, StudentName, IssueDate FROM IssuedBooks;

   
    DECLARE CONTINUE HANDLER FOR NOT FOUND SET done = 1;

   
    OPEN cur;

    read_loop: LOOP
        FETCH cur INTO b_id, s_name, i_date;

        IF done = 1 THEN
            LEAVE read_loop;
        END IF;

        SELECT b_id AS BookID, s_name AS StudentName, i_date AS IssueDate;

    END LOOP;
Close cursor
    CLOSE cur;
END $$

DELIMITER ;
 Step 5: Execute Stored Procedure
CALL ShowIssuedBooks();

-- 4. TRIGGER: CHECK BALANCE BEFORE WITHDRAWAL
DELIMITER $$

CREATE TRIGGER check_balance_before_withdrawal
BEFORE INSERT ON Transactions
FOR EACH ROW
BEGIN
    DECLARE current_balance DECIMAL(10,2);

    SELECT Balance INTO current_balance
    FROM Accounts
    WHERE AccountID = NEW.AccountID;

    IF NEW.TransactionType = 'Withdrawal' AND NEW.Amount > current_balance THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Insufficient Balance! Withdrawal not allowed';
    END IF;
END$$

DELIMITER ;


-- 5. TRIGGER: UPDATE BALANCE AFTER TRANSACTION
DELIMITER $$

CREATE TRIGGER update_balance_after_transaction
AFTER INSERT ON Transactions
FOR EACH ROW
BEGIN
    IF NEW.TransactionType = 'Deposit' THEN
        UPDATE Accounts
        SET Balance = Balance + NEW.Amount
        WHERE AccountID = NEW.AccountID;

    ELSEIF NEW.TransactionType = 'Withdrawal' THEN
        UPDATE Accounts
        SET Balance = Balance - NEW.Amount
        WHERE AccountID = NEW.AccountID;
    END IF;
END$$

DELIMITER ;
